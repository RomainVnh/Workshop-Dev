1. MODÈLE DE DONNÉES (Exemple) :
--------------------------------------------------------------------------------


users
  id, email (unique), password, username (unique), display_name,
  avatar_path, bio, city, postal_code, country,
  tolerance_level (enum: 'jumelle_exacte' | 'assortie' | 'tout_accepte'),
  trades_count, is_active, last_login, created, modified
  

socks                         (les fiches — le coeur du projet)
  id, user_id (FK), title, description,
  foot (enum: 'gauche'|'droite'|'indifferent'),
  size_eu (int) ou size_range (enum: '18-22','23-26',... 'taille unique'),
  color_primary_id (FK colors), color_secondary_id (FK colors, nullable),
  pattern (enum: 'uni','rayures','pois','jacquard','chevrons','motif_fantaisie',
           'sport','chaussette_de_noel','autre'),
  material (enum: 'coton','laine','laine_merinos','bambou','synthetique',
            'cachemire','mystere'),
  wear_level (enum: 'neuve','bon_etat','vecue','heroique'),   
  height (enum: 'socquette','mi-mollet','haute','montante'),
  status (enum: 'disponible','reservee','troquee','retiree'),
  lost_since (date, nullable)  <- "orpheline depuis le..."
  views_count, created, modified
  INDEX sur (status, color_primary_id, size_eu, pattern) pour les filtres


sock_photos
  id, sock_id (FK), path, is_cover (bool), position (int), created
  -> 1 à 4 photos par fiche, une seule cover


colors                        (table de référence, seedée)
  id, name ('bleu marine'), slug, hex ('#1b2a4a'), family ('bleu')
  -> permet un filtre par famille de couleur ET un rendu pastille


trades                        (les propositions de troc)
  id, requester_id (FK users), owner_id (FK users),
  target_sock_id (FK socks)          <- la chaussette convoitée
  status (enum: 'en_attente','acceptee','refusee','annulee','finalisee','expiree'),
  message (text, le mot d'accroche)
  responded_at, completed_at, expires_at, created, modified


trade_items                   (ce que le demandeur met sur la table)
  id, trade_id (FK), sock_id (FK)
  -> N chaussettes proposées en échange (souvent 1, parfois plusieurs au choix)

messages                      (fil de discussion adossé au troc)
  id, trade_id (FK), user_id (FK), body (text), read_at, created

favorites                     (mise de côté)
  id, user_id (FK), sock_id (FK), created   UNIQUE(user_id, sock_id)

reports                       (signalements)
  id, reporter_id (FK), sock_id ou user_id, reason, comment, status, created

badges + users_badges
  badges: id, slug, name, description, icon, threshold_type, threshold_value
  users_badges: user_id, badge_id, awarded_at

reviews                       (V1 — évaluation après troc finalisé)
  id, trade_id (FK), author_id, target_user_id, rating (1-5), comment, created
